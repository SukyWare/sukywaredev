local FOLDER_NAME = "SukyWare_Data"
local KEYS_FOLDER = FOLDER_NAME .. "/Keys"
local SESSION_FILE = FOLDER_NAME .. "/Session_" .. tostring(Player.UserId) .. ".json"
local WEBHOOK_COOLDOWN = 8 -- segundos entre envios de webhook por tentativa repetida

local lastWebhookSent = 0
local lastWebhookSignature = ""
local remoteData = { binds = {}, revoked = {} }

local function ensureFolders()
    if makefolder then
        pcall(function()
            if not isfolder(FOLDER_NAME) then makefolder(FOLDER_NAME) end
            if not isfolder(KEYS_FOLDER) then makefolder(KEYS_FOLDER) end
        end)
    end
end

local function normalizeKey(raw)
    if type(raw) ~= "string" then return "" end
    local clean = raw:match("^%s*(.-)%s*$") or ""
    return string.upper(clean)
end

local function getExecutorName()
    if identifyexecutor then
        local ok, name = pcall(identifyexecutor)
        if ok and name and tostring(name) ~= "" then return tostring(name) end
    end
    if getexecutorname then
        local ok, name = pcall(getexecutorname)
        if ok and name and tostring(name) ~= "" then return tostring(name) end
    end
    return "Desconhecido"
end

local function getPlatformName()
    if UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled then
        return "Mobile / Touch"
    elseif UserInputService.GamepadEnabled and not UserInputService.KeyboardEnabled then
        return "Console / Gamepad"
    end
    return "PC / Desktop"
end

local function getGameName()
    local ok, info = pcall(function()
        return MarketplaceService:GetProductInfoAsync(game.PlaceId)
    end)
    if ok and info and info.Name then
        return info.Name
    end
    return "PlaceId " .. tostring(game.PlaceId)
end

local function getAccountAgeDays()
    return tonumber(Player.AccountAge) or 0
end

local function getPingMs()
    local ok, ping = pcall(function()
        return math.floor(Player:GetNetworkPing() * 1000)
    end)
    return ok and ping or 0
end

local function getFpsEstimate()
    local dt = RunService.RenderStepped:Wait()
    return dt > 0 and math.floor(1 / dt) or 60
end

local function fetchRemoteSync()
    if REMOTE_SYNC_URL == "" then return end
    pcall(function()
        local raw = game:HttpGet(REMOTE_SYNC_URL)
        local decoded = HttpService:JSONDecode(raw)
        if type(decoded) == "table" then
            if type(decoded.binds) == "table" then
                remoteData.binds = decoded.binds
            end
            if type(decoded.revoked) == "table" then
                remoteData.revoked = decoded.revoked
            end
        end
    end)
end

local function isKeyRevoked(key)
    for _, rev in ipairs(remoteData.revoked or {}) do
        if normalizeKey(rev) == key then
            return true
        end
    end
    return false
end

local function sendWebhook(keyUsed, statusText, colorHex, extraInfo)
    if not WEBHOOK_URL or WEBHOOK_URL == "" then return end

    local signature = tostring(keyUsed) .. "|" .. tostring(statusText) .. "|" .. tostring(Player.UserId)
    local now = os.clock()
    if signature == lastWebhookSignature and (now - lastWebhookSent) < WEBHOOK_COOLDOWN then
        return
    end
    lastWebhookSent = now
    lastWebhookSignature = signature

    local req = (syn and syn.request) or (http and http.request) or http_request or request
    if not req then return end

    local executor = getExecutorName()
    local gameName = getGameName()
    local platform = getPlatformName()
    local ageDays = getAccountAgeDays()
    local pingMs = getPingMs()
    local fpsEst = getFpsEstimate()

    local fields = {
        {
            name = "Key usada:",
            value = "`" .. tostring(keyUsed) .. "`",
            inline = true
        },
        {
            name = "ID do usuário:",
            value = "`" .. tostring(Player.UserId) .. "` (@" .. tostring(Player.Name) .. ")",
            inline = true
        },
        {
            name = "Status:",
            value = tostring(statusText),
            inline = true
        },
        {
            name = "Executor:",
            value = "`" .. executor .. "`",
            inline = true
        },
        {
            name = "Plataforma:",
            value = "`" .. platform .. "`",
            inline = true
        },
        {
            name = "Idade da conta / Ping / FPS:",
            value = "`" .. tostring(ageDays) .. "d` • `" .. tostring(pingMs) .. "ms` • `~" .. tostring(fpsEst) .. " FPS`",
            inline = true
        },
        {
            name = "Jogo:",
            value = gameName .. " (`" .. tostring(game.PlaceId) .. "`)\nJobId: `" .. tostring(game.JobId) .. "`",
            inline = false
        }
    }

    if extraInfo and extraInfo ~= "" then
        table.insert(fields, {
            name = "Detalhes adicionais:",
            value = tostring(extraInfo),
            inline = false
        })
    end

    local payload = {
        embeds = {
            {
                title = "SukyWare • Key System Log",
                color = colorHex or 3447003,
                thumbnail = {
                    url = "https://www.roblox.com/headshot-thumbnail/image?userId=" .. tostring(Player.UserId) .. "&width=150&height=150&format=png"
                },
                fields = fields,
                footer = {
                    text = "SukyWare Security • UserId Lock & Telemetry"
                },
                timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
            }
        }
    }

    pcall(function()
        req({
            Url = WEBHOOK_URL,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json"
            },
            Body = HttpService:JSONEncode(payload)
        })
    end)
end

local function readKeyFile(key)
    if not (isfile and readfile) then return nil end
    local path = KEYS_FOLDER .. "/" .. key .. ".json"
    if not isfile(path) then return nil end

    local ok, raw = pcall(readfile, path)
    if not ok or type(raw) ~= "string" then return nil end

    local okJson, decoded = pcall(function()
        return HttpService:JSONDecode(raw)
    end)
    if okJson and type(decoded) == "table" then
        return decoded
    end
    return nil
end

local function writeKeyFile(key, dataTable)
    if not writefile then return false end
    ensureFolders()
    local path = KEYS_FOLDER .. "/" .. key .. ".json"
    local ok = pcall(function()
        writefile(path, HttpService:JSONEncode(dataTable))
    end)
    return ok
end

local function saveAutoSession(key, autoLoginEnabled)
    if not writefile then return end
    ensureFolders()
    pcall(function()
        writefile(SESSION_FILE, HttpService:JSONEncode({
            UserId = Player.UserId,
            Key = key,
            AutoLogin = autoLoginEnabled,
            UpdatedAt = os.time()
        }))
    end)
end

local function loadAutoSession()
    if not (isfile and readfile) then return nil end
    if not isfile(SESSION_FILE) then return nil end
    local ok, raw = pcall(readfile, SESSION_FILE)
    if not ok or type(raw) ~= "string" then return nil end
    local okJson, decoded = pcall(function()
        return HttpService:JSONDecode(raw)
    end)
    if okJson and type(decoded) == "table" and tonumber(decoded.UserId) == Player.UserId then
        return decoded
    end
    return nil
end

local function isKeyValid(input)
    if isKeyRevoked(input) then return false, "Revoked" end
    if ValidKeySet[input] then return true, "Valid" end
    return false, "Invalid"
end

local function verifyAndLockUserId(enteredKey)
    -- 1. Checa bind remoto se configurado
    local remoteBoundId = remoteData.binds and remoteData.binds[enteredKey]
    if remoteBoundId and tonumber(remoteBoundId) ~= Player.UserId then
        return "LockedByAnother", tonumber(remoteBoundId)
    end

    -- 2. Checa bind local por key
    local existing = readKeyFile(enteredKey)
    if existing and existing.UserId then
        local boundId = tonumber(existing.UserId)
        if boundId == Player.UserId then
            existing.LastUsed = os.time()
            existing.Uses = (tonumber(existing.Uses) or 1) + 1
            existing.UserName = Player.Name
            writeKeyFile(enteredKey, existing)
            return "Granted", boundId
        else
            return "LockedByAnother", boundId
        end
    end

    -- 3. Vincula na primeira vez
    local record = {
        Key = enteredKey,
        UserId = Player.UserId,
        UserName = Player.Name,
        FirstBound = os.time(),
        LastUsed = os.time(),
        Uses = 1,
        Executor = getExecutorName()
    }
    writeKeyFile(enteredKey, record)
    return "GrantedFirstTime", Player.UserId
end

local hasLoadedMain = false
local function launchMainScript()
    if hasLoadedMain then return end
    hasLoadedMain = true

    task.spawn(function()
        local ok, err = pcall(function()
            local source = game:HttpGet(SCRIPT_URL)
            local fn = loadstring(source)
            if type(fn) == "function" then
                fn()
            else
                error("Conteúdo do Script.txt inválido ou vazio.")
            end
        end)
        if not ok then
            warn("[SukyWare] Erro ao carregar Script.txt: " .. tostring(err))
        end
    end)
end

local isSubmitting = false
local function setButtonFeedback(text, color, duration)
    SubmitButton.Text = text
    SubmitButton.BackgroundColor3 = color
    task.wait(duration or 1.3)
    SubmitButton.Text = "Submit Key  >"
    SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 125, 255)
end

local function processKey(rawInput, isAutoAttempt)
    if isSubmitting then return end
    isSubmitting = true

    local enteredKey = normalizeKey(rawInput)
    KeyBox.Text = enteredKey

    if enteredKey == "" then
        setButtonFeedback("Digite uma Key!", Color3.fromRGB(220, 150, 50), 1.2)
        isSubmitting = false
        return
    end

    local valid, reason = isKeyValid(enteredKey)
    if not valid then
        local statusMsg = (reason == "Revoked") and "Key revogada / bloqueada" or "Tentativa com key inválida"
        task.spawn(function()
            sendWebhook(enteredKey, statusMsg, 15105570, isAutoAttempt and "Tentativa via Auto-Login" or nil)
        end)
        setButtonFeedback(reason == "Revoked" and "Key Revogada!" or "Key Incorreta!", Color3.fromRGB(220, 55, 65), 1.3)
        isSubmitting = false
        return
    end

    local lockStatus, boundUserId = verifyAndLockUserId(enteredKey)

    if lockStatus == "GrantedFirstTime" then
        if autoLoginEnabled then
            saveAutoSession(enteredKey, true)
        end
        task.spawn(function()
            sendWebhook(enteredKey, "Vinculada ao UserId e liberada (1º uso)", 3066993)
        end)
        SubmitButton.Text = "Key Vinculada & Liberada!"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 195, 105)
        task.wait(0.6)
        closeGui()
        launchMainScript()

    elseif lockStatus == "Granted" then
        if autoLoginEnabled then
            saveAutoSession(enteredKey, true)
        end
        task.spawn(function()
            sendWebhook(
                enteredKey,
                isAutoAttempt and "Acesso liberado via Auto-Login" or "Acesso liberado (UserId confirmado)",
                3447003
            )
        end)
        SubmitButton.Text = "Acesso Liberado!"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 195, 105)
        task.wait(0.5)
        closeGui()
        launchMainScript()

    elseif lockStatus == "LockedByAnother" then
        task.spawn(function()
            sendWebhook(
                enteredKey,
                "Bloqueada — pertence a outro UserId",
                15158332,
                "Dono original UserId: `" .. tostring(boundUserId) .. "`"
            )
        end)
        setButtonFeedback("Key vinculada a outro ID!", Color3.fromRGB(220, 55, 65), 2)
        isSubmitting = false
    else
        isSubmitting = false
    end
end

SubmitButton.MouseButton1Click:Connect(function()
    processKey(KeyBox.Text, false)
end)

    if enterPressed then
        processKey(KeyBox.Text, false)
