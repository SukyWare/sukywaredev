local DIR = "SukyWare_Data"
local SESSION_FILE = DIR .. "/CloudSession_" .. tostring(Player.UserId) .. ".json"

local function norm(s)
    if type(s) ~= "string" then return "" end
    return string.upper(s:match("^%s*(.-)%s*$") or "")
end

local function getExec()
    for _, fn in ipairs({ identifyexecutor, getexecutorname }) do
        if fn then
            local ok, n = pcall(fn)
            if ok and n and tostring(n) ~= "" then return tostring(n) end
        end
    end
    return "Desconhecido"
end

local function getGame()
    local ok, info = pcall(function() return MarketplaceService:GetProductInfoAsync(game.PlaceId) end)
    return (ok and info and info.Name) or ("PlaceId " .. tostring(game.PlaceId))
end

local function readSession()
    if not (isfile and readfile) or not isfile(SESSION_FILE) then return nil end
    local ok, raw = pcall(readfile, SESSION_FILE)
    if not ok or type(raw) ~= "string" then return nil end
    local okJ, d = pcall(function() return HttpService:JSONDecode(raw) end)
    return (okJ and type(d) == "table" and tonumber(d.UserId) == Player.UserId) and d or nil
end

local function saveSession(key, auto)
    if not writefile then return end
    if makefolder and isfolder and not isfolder(DIR) then pcall(makefolder, DIR) end
    pcall(function()
        writefile(SESSION_FILE, HttpService:JSONEncode({
            UserId = Player.UserId,
            Key = key,
            AutoLogin = auto
        }))
    end)
end

local loaded, busy = false, false

local function runHub(remoteScript)
    if loaded then return end
    loaded = true
    task.spawn(function()
        local ok, err = pcall(function()
            local src = (type(remoteScript) == "string" and remoteScript ~= "") and remoteScript or game:HttpGet(FALLBACK_SCRIPT)
            loadstring(src)()
        end)
        if not ok then warn("[SukyWare] Erro ao carregar hub: " .. tostring(err)) end
    end)
end

local function flashBtn(txt, col, waitS)
    submitBtn.Text = txt
    submitBtn.BackgroundColor3 = col
    task.wait(waitS or 1.4)
    submitBtn.Text = "Submit Key >"
    submitBtn.BackgroundColor3 = Color3.fromRGB(0, 125, 255)
end

local function verifyKey(raw)
    if busy then return end
    busy = true

    local k = norm(raw)
    keyBox.Text = k

    if k == "" then
        flashBtn("Digite uma Key!", Color3.fromRGB(220, 150, 50), 1.1)
        busy = false
        return
    end

    local req = (syn and syn.request) or (http and http.request) or http_request or request
    if not req then
        flashBtn("Executor sem HTTP!", Color3.fromRGB(220, 55, 65), 1.8)
        busy = false
        return
    end

    submitBtn.Text = "Verificando..."
    submitBtn.BackgroundColor3 = Color3.fromRGB(35, 95, 190)

    local okReq, res = pcall(function()
        return req({
            Url = API_URL .. "/verify",
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode({
                key = k,
                userId = tostring(Player.UserId),
                userName = Player.Name,
                executor = getExec(),
                gameName = getGame(),
                placeId = tostring(game.PlaceId)
            })
        })
    end)

    if not okReq or not res or not res.Body then
        flashBtn("Erro de conexão!", Color3.fromRGB(220, 55, 65), 1.6)
        busy = false
        return
    end

    local okJ, data = pcall(function() return HttpService:JSONDecode(res.Body) end)
    if not okJ or type(data) ~= "table" then
        flashBtn("Erro na API!", Color3.fromRGB(220, 55, 65), 1.6)
        busy = false
        return
    end

    if data.valid == true then
        if autoLogin then saveSession(k, true) end
        submitBtn.Text = (data.plan and data.remaining) and (data.plan .. " (" .. data.remaining .. ")") or "Key Correta!"
        submitBtn.BackgroundColor3 = Color3.fromRGB(0, 195, 105)
        task.wait(0.7)
        sg:Destroy()
        runHub(data.script)
    else
        flashBtn(tostring(data.message or "Key Incorreta!"), Color3.fromRGB(220, 55, 65), 1.6)
        busy = false
    end
end
