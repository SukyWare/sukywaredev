-- Função genérica para disparar webhooks
local function SendWebhook(url, dataTable)
    if not url or url == "" then return end
    local encoded = HttpService:JSONEncode(dataTable)
    local requestFunc = syn and syn.request or http_request or request or HttpPost
    if requestFunc then
        pcall(function()
            requestFunc({
                Url = url,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = encoded
            })
        end)
    end
end

-- Função de validação da Key na lista
local function isKeyValid(input)
    for _, key in ipairs(ValidKeys) do
        if key == input then
            return true
        end
    end
    return false
end

-- Sistema de Trava por ID (UserId Lock via Arquivo Local)
local function verifyAndLockUserId(enteredKey)
    local folderName = "SukyWareV2_Data"
    local fileName = folderName .. "/" .. enteredKey .. ".json"
    
    if makefolder and not isfolder(folderName) then
        pcall(function() makefolder(folderName) end)
    end
    
    if isfile and isfile(fileName) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(fileName))
        end)
        
        if success and data and data.UserId then
            if data.UserId == Player.UserId then
                return "Granted" -- Mesmo dono entrando de novo
            else
                return "LockedByAnother" -- Outro ID tentando usar
            end
        end
    else
        if writefile then
            pcall(function()
                local saveData = {
                    UserId = Player.UserId,
                    UserName = Player.Name,
                    ActivationDate = os.date("%Y-%m-%d %H:%M:%S")
                }
                writefile(fileName, HttpService:JSONEncode(saveData))
            end)
        end
        return "GrantedFirstTime" -- Primeira ativação desta key
    end
    
    return "Error"
end

-- Ação do Botão Submit
SubmitButton.MouseButton1Click:Connect(function()
    local enteredKey = KeyBox.Text
    
    if enteredKey == "" then
        SubmitButton.Text = "Digite uma Key!"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(220, 150, 50)
        task.wait(1.2)
        SubmitButton.Text = "Submit Key >"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        return
    end
    
    if isKeyValid(enteredKey) then
        local lockStatus = verifyAndLockUserId(enteredKey)
        
        if lockStatus == "Granted" or lockStatus == "GrantedFirstTime" then
            SubmitButton.Text = "Key Correta!"
            SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
            
            -- Dispara as mensagens no Discord
            task.spawn(function()
                -- 1. Sempre manda o log para o canal de logs gerais
                SendSecurityLog(
                    "✅ Acesso Concedido - SukyWare V2", 
                    65280, 
                    "O usuário ativou/entrou com sucesso utilizando uma Key válida.", 
                    enteredKey
                )
                
                -- 2. Se for a PRIMEIRA vez que essa key está sendo ativada (resgate), manda no canal de resgates
                if lockStatus == "GrantedFirstTime" then
                    SendKeyRedemptionLog(enteredKey)
                end
            end)
            
            -- Destroi a janela de key imediatamente
            ScreenGui:Destroy()
            
            -- Executa o seu loadstring do GitHub
            task.spawn(function()
                local success, err = pcall(function()
                    loadstring(game:HttpGet("https://raw.githubusercontent.com/SukyWare/Sukywarev1/refs/heads/main/Script.txt"))()
                end)
                if not success then
                    warn("Erro ao carregar o script principal: " .. tostring(err))
                end
            end)
            
        elseif lockStatus == "LockedByAnother" then
            SubmitButton.Text = "Key vinculada a outro ID!"
            SubmitButton.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
            
            -- Envia log de tentativa de uso duplicado apenas para o canal de logs
            task.spawn(function()
                SendSecurityLog(
                    "⛔ Tentativa de Uso Duplicado (Bloqueado)", 
                    16711680, 
                    "Este usuário tentou usar uma Key que já está trancada em outra conta do Roblox!", 
                    enteredKey
                )
            end)
            
            task.wait(2)
            SubmitButton.Text = "Submit Key >"
            SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        end
        
    else
        SubmitButton.Text = "Key Incorreta!"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
        
        -- Envia log de key inválida apenas para o canal de logs
        task.spawn(function()
            SendSecurityLog(
                "❌ Key Inválida Tentada", 
                16753920, 
                "O usuário digitou uma chave que não existe na lista de keys válidas.", 
                enteredKey
            )
        end)
        
        task.wait(1.2)
        SubmitButton.Text = "Submit Key >"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
    end
end)
