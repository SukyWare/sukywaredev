local function isKeyValid(input)
    for _, key in ipairs(ValidKeys) do
        if key == input then
            return true
        end
    end
    return false
end

local function verifyAndLockUserId(enteredKey)
    local folderName = "SukyWare_Data"
    local fileName = folderName .. "/" .. enteredKey .. ".json"
    
    if makefolder and not isfolder(folderName) then
        pcall(function() makefolder(folderName) end)
    end
    
    if isfile and isfile(fileName) then
        local success, data = pcall(function()
            return HttpService:JSONDecode(readfile(fileName))
        end)
        
        if success and data and data.UserId then
            if data.UserId == Player.UserId then
                return "Granted"
            else
                return "LockedByAnother"
            end
        end
    else
        if writefile then
            pcall(function()
                local saveData = {
                    UserId = Player.UserId,
                    UserName = Player.Name
                }
                writefile(fileName, HttpService:JSONEncode(saveData))
            end)
        end
        return "GrantedFirstTime"
    end
    
    return "Granted"
end

SubmitButton.MouseButton1Click:Connect(function()
    local enteredKey = KeyBox.Text
    
    if enteredKey == "" then
        SubmitButton.Text = "Digite uma Key!"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(220, 150, 50)
        task.wait(1.2)
        SubmitButton.Text = "Submit Key >"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        return
    end
    
    if isKeyValid(enteredKey) then
        local lockStatus = verifyAndLockUserId(enteredKey)
        
        if lockStatus == "Granted" or lockStatus == "GrantedFirstTime" then
            SubmitButton.Text = "Key Correta!"
            SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 200, 100)
            task.wait(0.5)
            ScreenGui:Destroy()
            
            task.spawn(function()
                local success, err = pcall(function()
                    loadstring(game:HttpGet("https://raw.githubusercontent.com/SukyWare/sukywaredevv2/refs/heads/main/Script.txt"))()
                end)
                if not success then
                    warn("Erro ao carregar: " .. tostring(err))
                end
            end)
            
        elseif lockStatus == "LockedByAnother" then
            SubmitButton.Text = "Key vinculada a outro ID!"
            SubmitButton.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
            task.wait(2)
            SubmitButton.Text = "Submit Key >"
            SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        end
    else
        SubmitButton.Text = "Key Incorreta!"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
        task.wait(1.2)
        SubmitButton.Text = "Submit Key >"
        SubmitButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
    end
end)
